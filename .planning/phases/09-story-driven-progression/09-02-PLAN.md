    const SEEN_KEY    = 'emma-story-seen-worlds'

      try {
        const v = localStorage.getItem(key)
        return v !== null ? JSON.parse(v) : fallback
      } catch { return fallback }
    }

    export function useStory() {
      const showCutscene   = ref(false)
      const cutsceneWorld  = ref(null)
      const seenWorlds     = ref(new Set(getJSON(SEEN_KEY, [])))
      const activePowerUp  = ref(getJSON(POWERUP_KEY, null))

      function buildWorld(worldNum, characterId) {
        const lines = (DIALOGUE_BY_CHAR[characterId] || GENERIC_DIALOGUE)[worldNum]
                   || GENERIC_DIALOGUE[worldNum]
        return {
          worldNum,
          dialogueLines: lines,
          powerUp: POWER_UPS[worldNum] || null,
          isFinale: worldNum === 7,
          characterId,
        }
      }

      function triggerCutscene(levelNum, characterId) {
        if (!POWER_UPS[levelNum]) return  // unknown world; ignore
        cutsceneWorld.value = buildWorld(levelNum, characterId)
        showCutscene.value  = true
      }

      function dismissCutscene() {
        const w = cutsceneWorld.value
        if (w) {
          // Mark seen
          seenWorlds.value = new Set([...seenWorlds.value, w.worldNum])
          setJSON(SEEN_KEY, [...seenWorlds.value])
          // Activate power-up
          activePowerUp.value = w.powerUp
          setJSON(POWERUP_KEY, w.powerUp)
        }
        showCutscene.value  = false
        cutsceneWorld.value = null
      }

      function resetStory() {
        seenWorlds.value    = new Set()
        activePowerUp.value = null
        removeKey(SEEN_KEY)
        removeKey(POWERUP_KEY)
      }

      // Convenience: consumed by App.vue power-up application logic in Plan 03
      const isDoubleStarsActive = computed(() => activePowerUp.value?.type === 'doubleStars')

      return {
        showCutscene,
        cutsceneWorld,
        seenWorlds,
        activePowerUp,
        isDoubleStarsActive,
        triggerCutscene,
        dismissCutscene,
        resetStory,
      }
    }

    Props:
      - `show`         (Boolean, default false)
      - `world`        (Object, default null) — shape: { worldNum, dialogueLines: string[], powerUp: {type,label,worldNum}, isFinale: boolean, characterId: string }

    Emits:

    Structure:
    ```vue
    <script setup>
    import { ref, computed, watch } from 'vue'

    const props = defineProps({
      show:  { type: Boolean, default: false },
      world: { type: Object,  default: null  },
    })
    const emit = defineEmits(['dismiss'])

    const lineIndex = ref(0)
    const isLastLine = computed(() => {
      if (!props.world) return false
      return lineIndex.value >= props.world.dialogueLines.length - 1
    })
    const currentLine = computed(() => props.world?.dialogueLines?.[lineIndex.value] ?? '')

    // Reset on every fresh open
    watch(() => props.show, (v) => { if (v) lineIndex.value = 0 })

    function advance() {
      if (isLastLine.value) {
        emit('dismiss')
      } else {
        lineIndex.value++
      }
    }
    </script>

    <template>
      <Transition name="cutscene-fade">
        <div
          v-if="show && world"
          class="fixed inset-0 z-[225] flex flex-col items-center justify-center bg-black/70 px-6"
          @click.self="advance"
        >
          <!-- World 7 finale variant: YOU WIN! banner -->
          <div
            v-if="world.isFinale"
            class="mb-6 text-5xl font-extrabold text-yellow-300 drop-shadow-lg animate-pulse"
          >YOU WIN!</div>

          <div class="max-w-md w-full rounded-2xl bg-white p-6 shadow-2xl text-center">
            <p class="text-xl text-gray-800 leading-relaxed min-h-[4rem]">
              {{ currentLine }}
            </p>

            <!-- Power-up reveal on the final line -->
            <div
              v-if="isLastLine && world.powerUp"
              class="mt-5 rounded-xl bg-yellow-100 border-2 border-yellow-400 p-3"
            >
              <p class="text-sm uppercase tracking-wide text-yellow-700">Power-up unlocked!</p>
              <p class="text-2xl font-bold text-yellow-900">{{ world.powerUp.label }}</p>
            </div>

            <button
              class="mt-6 w-full rounded-xl bg-red-500 hover:bg-red-600 active:bg-red-700 text-white font-bold py-3 px-6 text-lg shadow-md"
              @click="advance"
            >
              {{ isLastLine ? (world.isFinale ? 'PLAY AGAIN →' : "LET'S GO! →") : 'NEXT →' }}
            </button>
          </div>
        </div>
      </Transition>
    </template>

    <style scoped>
    .cutscene-fade-enter-active { transition: opacity 0.45s ease; }
    .cutscene-fade-leave-active { transition: opacity 0.30s ease; }
    .cutscene-fade-enter-from,
    .cutscene-fade-leave-to     { opacity: 0; }
    </style>
    ```
