    ```javascript
    import { describe, it, expect, beforeEach } from 'vitest'
    import { useStory } from '../useStory.js'

    describe('useStory', () => {
      beforeEach(() => {
        localStorage.clear()
      })

      describe('STORY-01a: world progression', () => {
        it('triggerCutscene(1, "peach") opens cutscene for world 1', () => {
          const s = useStory()
          s.triggerCutscene(1, 'peach')
          expect(s.showCutscene.value).toBe(true)
          expect(s.cutsceneWorld.value?.worldNum).toBe(1)
          expect(Array.isArray(s.cutsceneWorld.value?.dialogueLines)).toBe(true)
          expect(s.cutsceneWorld.value.dialogueLines.length).toBeGreaterThan(0)
        })

        it('cutsceneWorld is null before any trigger', () => {
          const s = useStory()
          expect(s.cutsceneWorld.value).toBeNull()
          expect(s.showCutscene.value).toBe(false)
        })
      })

      describe('STORY-01b: seenWorlds guard', () => {
        it('dismissCutscene marks world as seen and persists to localStorage', () => {
          const s = useStory()
          s.triggerCutscene(1, 'peach')
          s.dismissCutscene()
          expect(s.seenWorlds.value.has(1)).toBe(true)
          const stored = JSON.parse(localStorage.getItem('emma-story-seen-worlds'))
          expect(stored).toContain(1)
        })

        it('seenWorlds hydrates from localStorage on construction', () => {
          localStorage.setItem('emma-story-seen-worlds', JSON.stringify([1, 2, 3]))
          const s = useStory()
          expect(s.seenWorlds.value.has(1)).toBe(true)
          expect(s.seenWorlds.value.has(2)).toBe(true)
          expect(s.seenWorlds.value.has(3)).toBe(true)
        })
      })

      describe('STORY-01c: power-up grant on dismiss', () => {
        const expectedPowerUps = [
          [1, 'doubleStars'],
          [2, 'timeFreeze'],
          [3, 'hint'],
          [4, 'skipQuestion'],
          [5, 'starShield'],
          [6, 'streakSaver'],
          [7, 'bonusRound'],
        ]
        for (const [world, type] of expectedPowerUps) {
          it(`world ${world} grants ${type} power-up`, () => {
            const s = useStory()
            s.triggerCutscene(world, 'peach')
            s.dismissCutscene()
            expect(s.activePowerUp.value?.type).toBe(type)
            expect(s.activePowerUp.value?.worldNum).toBe(world)
            const stored = JSON.parse(localStorage.getItem('emma-story-active-powerup'))
            expect(stored?.type).toBe(type)
          })
        }

        it('activePowerUp is null until a cutscene is dismissed', () => {
          const s = useStory()
          s.triggerCutscene(1, 'peach')
          expect(s.activePowerUp.value).toBeNull() // only granted on dismiss
        })
      })

      describe('STORY-01d: resetStory', () => {
        it('clears seenWorlds, activePowerUp, and localStorage', () => {
          const s = useStory()
          s.triggerCutscene(1, 'peach')
          s.dismissCutscene()
          s.resetStory()
          expect(s.activePowerUp.value).toBeNull()
          expect(s.seenWorlds.value.size).toBe(0)
          const seen = localStorage.getItem('emma-story-seen-worlds')
          const pu   = localStorage.getItem('emma-story-active-powerup')
          // Either removed or empty/null JSON — both acceptable
          expect(seen === null || seen === '[]').toBe(true)
          expect(pu === null || pu === 'null').toBe(true)
        })
      })

      describe('character-specific dialogue', () => {
        it('returns different dialogue lines for peach vs daisy on world 1', () => {
          const s1 = useStory()
          s1.triggerCutscene(1, 'peach')
          const peachLines = s1.cutsceneWorld.value.dialogueLines.join('|')

          const s2 = useStory()
          s2.triggerCutscene(1, 'daisy')
          const daisyLines = s2.cutsceneWorld.value.dialogueLines.join('|')

          expect(peachLines).not.toBe(daisyLines)
        })

        it('falls back to generic dialogue for Phase 8 characters (e.g. yoshi)', () => {
          const s = useStory()
          s.triggerCutscene(1, 'yoshi')
          expect(Array.isArray(s.cutsceneWorld.value?.dialogueLines)).toBe(true)
          expect(s.cutsceneWorld.value.dialogueLines.length).toBeGreaterThan(0)
        })
      })
    })
    ```
