```js
function isUnlocked(char) {
```

```js
function selectCharacter(char) {
  if (!isUnlocked(char)) return
  emit('select', char)
}
```

```html
  :class="[char.border, isUnlocked(char) ? 'hover:scale-105 cursor-pointer' : 'opacity-40 grayscale cursor-not-allowed']"
  @click="selectCharacter(char)"
  ...
>
```
