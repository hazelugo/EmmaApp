Create `src/utils/difficultyTheme.js` exporting TWO named functions:

```javascript
/**
 * Pure utility: map a numeric difficulty signal to a CSS theme class.
 * Thresholds (inclusive upper bound):
 *   <= 7  → 'theme-overworld'
 *   <= 12 → 'theme-underground'
 *   <= 17 → 'theme-desert'
 *   >  17 → 'theme-space'
 * @param {number} maxOperand
 * @returns {'theme-overworld'|'theme-underground'|'theme-desert'|'theme-space'}
 */
export function pickDifficultyTheme (maxOperand) {
  const n = Number(maxOperand) || 0
  if (n <= 7)  return 'theme-overworld'
  if (n <= 12) return 'theme-underground'
  if (n <= 17) return 'theme-desert'
  return 'theme-space'
}

/**
 * Pick theme based on the highest operand cap across all operators.
 * @param {Record<string, number>} maxOperandByOperator
 * @returns {string} theme class
 */
export function pickMaxDifficultyTheme (maxOperandByOperator) {
  if (!maxOperandByOperator) return pickDifficultyTheme(0)
  const values = Object.values(maxOperandByOperator).map(v => Number(v) || 0)
  const max = values.length ? Math.max(...values) : 0
  return pickDifficultyTheme(max)
}
```

```javascript
import { describe, it, expect } from 'vitest'
import { pickDifficultyTheme, pickMaxDifficultyTheme } from './difficultyTheme.js'

describe('pickDifficultyTheme', () => {
  it('returns theme-overworld for low difficulty (3-7)', () => {
    expect(pickDifficultyTheme(3)).toBe('theme-overworld')
    expect(pickDifficultyTheme(7)).toBe('theme-overworld')
  })
  it('returns theme-underground for mid-low (8-12)', () => {
    expect(pickDifficultyTheme(8)).toBe('theme-underground')
    expect(pickDifficultyTheme(12)).toBe('theme-underground')
  })
  it('returns theme-desert for mid-high (13-17)', () => {
    expect(pickDifficultyTheme(13)).toBe('theme-desert')
    expect(pickDifficultyTheme(17)).toBe('theme-desert')
  })
  it('returns theme-space for high (18+)', () => {
    expect(pickDifficultyTheme(18)).toBe('theme-space')
    expect(pickDifficultyTheme(20)).toBe('theme-space')
  })
  it('handles 0 / undefined / non-number gracefully', () => {
    expect(pickDifficultyTheme(0)).toBe('theme-overworld')
    expect(pickDifficultyTheme(undefined)).toBe('theme-overworld')
    expect(pickDifficultyTheme(NaN)).toBe('theme-overworld')
  })
})

describe('pickMaxDifficultyTheme', () => {
  it('picks theme based on max across operators', () => {
    expect(pickMaxDifficultyTheme({ '+': 5, '-': 10, '×': 3, '÷': 3 })).toBe('theme-underground')
    expect(pickMaxDifficultyTheme({ '+': 20, '-': 20, '×': 10, '÷': 10 })).toBe('theme-space')
    expect(pickMaxDifficultyTheme({ '+': 14, '-': 6, '×': 3, '÷': 3 })).toBe('theme-desert')
  })
  it('returns theme-overworld for empty / null / undefined input', () => {
    expect(pickMaxDifficultyTheme({})).toBe('theme-overworld')
    expect(pickMaxDifficultyTheme(null)).toBe('theme-overworld')
    expect(pickMaxDifficultyTheme(undefined)).toBe('theme-overworld')
  })
})
```
