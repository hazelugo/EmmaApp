    ```javascript
    import { ref, onMounted } from 'vue'
    const emit = defineEmits(['result', 'skip'])

    const SHAPES = ['circle', 'square', 'triangle', 'diamond']
    const TOTAL_ROUNDS = 6
    const round = ref(1)
    const target = ref(null)
    const options = ref([])      // array of 4 shape strings, randomized
    const correct = ref(0)
    const feedback = ref(null)   // { index: number, kind: 'correct' | 'wrong' } | null
    const locked = ref(false)    // prevent double-tap during feedback

    function shuffle(arr) {
      const a = arr.slice()
      for (let i = a.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1))
        ;[a[i], a[j]] = [a[j], a[i]]
      }
      return a
    }

    function startRound() {
      target.value = SHAPES[Math.floor(Math.random() * SHAPES.length)]
      options.value = shuffle(SHAPES).slice(0, 4)   // SHAPES has exactly 4, so all distinct; target is guaranteed present
      feedback.value = null
      locked.value = false
    }

    function pick(idx) {
      if (locked.value) return
      locked.value = true
      const isCorrect = options.value[idx] === target.value
      if (isCorrect) correct.value++
      feedback.value = { index: idx, kind: isCorrect ? 'correct' : 'wrong' }
      setTimeout(() => {
        if (round.value >= TOTAL_ROUNDS) endGame()
        else { round.value++; startRound() }
      }, 400)
    }

    function endGame() {
      const won = correct.value >= 4
      emit('result', { won, bonus: correct.value })
    }

    function onSkip() { emit('skip') }

    onMounted(startRound)
    ```

    - 3 rounds. Round 1 sequence length = 3, Round 2 = 4, Round 3 = 5.

    - Phase 2 (input): banner changes to "Your turn!". Player taps colors in order. Each correct tap advances. First wrong tap = round fails.
    - End of round 3 OR first failure → end game.
    - Win condition: completed all 3 rounds (no failures).
    - Bonus: 3 coins on win, 0 on loss.

    UI:
    - Header: "Jump Sequence!" in `text-star-gold text-3xl font-bold`.
    - Round indicator: "Round N of 3".
    - Banner ("Watch carefully!" / "Your turn!").
    - 2x2 grid of buttons, each ~`w-24 h-24 rounded-2xl`. Active state during playback: `scale-110 ring-4 ring-white`.
    - Disable button taps during playback (`pointer-events-none opacity-70`).

    Skip button per D-03: bottom-right, small, neutral. Emits `skip`.

    Script:
    ```javascript
    import { ref, onMounted, onUnmounted } from 'vue'
    const emit = defineEmits(['result', 'skip'])

    const ROUND_LENGTHS = [3, 4, 5]
    const round = ref(1)
    const sequence = ref([])         // array of color indices 0..3
    const inputIndex = ref(0)
    const activeLight = ref(null)
    const phase = ref('playback')    // 'playback' | 'input' | 'done'
    const _timeouts = []

    function _clearTimeouts() {
      while (_timeouts.length) clearTimeout(_timeouts.shift())
      activeLight.value = null
    }

    function generateSequence(len) {
      const s = []
      for (let i = 0; i < len; i++) s.push(Math.floor(Math.random() * 4))
      return s
    }

    function playSequence() {
      phase.value = 'playback'
      sequence.value.forEach((colorIdx, i) => {
        _timeouts.push(setTimeout(() => { activeLight.value = colorIdx }, i * 600))
        _timeouts.push(setTimeout(() => { activeLight.value = null }, i * 600 + 400))
      })
      _timeouts.push(setTimeout(() => {
        phase.value = 'input'
        inputIndex.value = 0
      }, sequence.value.length * 600 + 200))
    }

    function startRound() {
      _clearTimeouts()
      sequence.value = generateSequence(ROUND_LENGTHS[round.value - 1])
      playSequence()
    }

    function tap(colorIdx) {
      if (phase.value !== 'input') return
      if (sequence.value[inputIndex.value] === colorIdx) {
        inputIndex.value++
        if (inputIndex.value >= sequence.value.length) {
          // round complete
          if (round.value >= ROUND_LENGTHS.length) endGame(true)
          else { round.value++; startRound() }
        }
      } else {
        endGame(false)
      }
    }

    function endGame(won) {
      phase.value = 'done'
      _clearTimeouts()
      emit('result', { won, bonus: won ? 3 : 0 })
    }

    function onSkip() { _clearTimeouts(); emit('skip') }

    onMounted(startRound)
    onUnmounted(_clearTimeouts)   // critical: prevent timer leak (Pitfall 2)
    ```

    Render the 4 buttons by mapping over `[0,1,2,3]` with colors mapped to Tailwind bg classes:
    - 0 → `bg-mario-red`, 1 → `bg-star-gold`, 2 → `bg-luigi`, 3 → `bg-rosalina`

    No localStorage. No star mutation.
