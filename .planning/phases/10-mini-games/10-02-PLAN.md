    ```javascript
    // Phase 10 — Mini-game orchestrator.
    // Per D-01: ABSOLUTE THRESHOLD, never modulo. Streak milestones at 5/10/15 already fire confetti.
    // Per D-02: This composable does NOT mutate stars. App.vue calls creditTimerCoins() on result.
    import { ref } from 'vue'

    export const MINI_GAME_TYPES = ['math-dash', 'shape-match', 'jump-sequence']

    function _randomThreshold(base = 0) {
      // 5, 6, or 7 inclusive. Returned as an absolute target (base + 5..7).
      return base + 5 + Math.floor(Math.random() * 3)
    }

    function _pickGame() {
      return MINI_GAME_TYPES[Math.floor(Math.random() * MINI_GAME_TYPES.length)]
    }

    export function useMiniGame() {
      const showMiniGame   = ref(false)
      const activeGame     = ref(null)
      const miniGameResult = ref(null)
      let   _nextThreshold = _randomThreshold(0)

      function checkMiniGameTrigger(streak, showLevelVictory = false) {
        if (showLevelVictory) return false              // Pitfall 5 from RESEARCH.md
        if (streak < _nextThreshold) return false
        // Trigger
        activeGame.value     = _pickGame()
        showMiniGame.value   = true
        miniGameResult.value = null
        _nextThreshold       = _randomThreshold(streak) // D-01: reset to streak + random(5,7)
        return true
      }

      function onMiniGameResult(result) {
        miniGameResult.value = result
        showMiniGame.value   = false
        activeGame.value     = null
      }

      function resetMiniGame() {
        showMiniGame.value   = false
        activeGame.value     = null
        miniGameResult.value = null
        _nextThreshold       = _randomThreshold(0)      // fresh 5–7 from zero
      }

      return {
        showMiniGame,
        activeGame,
        miniGameResult,
        checkMiniGameTrigger,
        onMiniGameResult,
        resetMiniGame,
      }
    }
    ```

    Do NOT add modulo arithmetic. Do NOT mutate stars. Do NOT touch localStorage (D-04).

    **File 2: `src/components/MathDashGame.vue`** — 15-second math sprint.

    Game rules (from RESEARCH.md "Math Dash"):
    - Operands range 1–10, operator is `+` or `-` chosen randomly per problem
    - 15-second countdown (1s interval, starts on mount)
    - Win condition: score >= 5 correct answers when timer ends
    - Bonus on win: `Math.floor(score / 2)` coins; on loss: 0
    - On end (timer hits 0 or user runs out of time), emit `result` with `{ won: boolean, bonus: number }`

    UI requirements:
    - Header line: "Math Dash!" in `text-star-gold` with `text-3xl font-bold`
    - Timer pill (top): shows `timeLeft`s remaining; turns red when timeLeft <= 5
    - Problem display centered, large (`text-5xl`): e.g., "7 + 4 = ?"
    - Inline numeric tap row (digits 0–9 in a single horizontal grid, 5x2 wrap) — NOT NumberPad component (per RESEARCH.md, inline is simpler)
    - Current typed answer shown above tap row in box; max 3 chars (V5 input cap per RESEARCH.md Security section)
    - Submit button below tap row; clears answer on submit and increments score if correct
    - Backspace button on tap row
    - Score counter shown bottom-left: "Score: N"
    - **Skip button per D-03**: positioned `absolute bottom-4 right-4`, small (`text-sm px-2 py-1`), neutral color (`text-white/60 hover:text-white/90 border border-white/20 rounded`), label "Skip". Click emits `skip` event (no payload — parent shell converts to `{ won: false, bonus: 0 }`).

    Script (Composition API, `<script setup>`):
    ```javascript
    import { ref, onMounted, onUnmounted } from 'vue'
    const emit = defineEmits(['result', 'skip'])

    const timeLeft = ref(15)
    const score    = ref(0)
    const answer   = ref('')
    const problem  = ref({ a: 0, b: 0, op: '+', solution: 0 })
    let _interval = null

    function nextProblem() {
      const a = 1 + Math.floor(Math.random() * 10)
      const b = 1 + Math.floor(Math.random() * 10)
      const op = Math.random() < 0.5 ? '+' : '-'
      // ensure non-negative result for subtraction
      const [x, y] = op === '-' && b > a ? [b, a] : [a, b]
      const solution = op === '+' ? x + y : x - y
      problem.value = { a: x, b: y, op, solution }
    }

    function appendDigit(d) {
      if (answer.value.length >= 3) return    // V5 input cap
      answer.value += String(d)
    }
    function backspace() { answer.value = answer.value.slice(0, -1) }

    function submit() {
      if (answer.value === '') return
      if (Number(answer.value) === problem.value.solution) score.value++
      answer.value = ''
      nextProblem()
    }

    function endGame() {
      if (_interval) { clearInterval(_interval); _interval = null }
      const won = score.value >= 5
      emit('result', { won, bonus: won ? Math.floor(score.value / 2) : 0 })
    }

    function onSkip() {
      if (_interval) { clearInterval(_interval); _interval = null }
      emit('skip')
    }

    onMounted(() => {
      nextProblem()
      _interval = setInterval(() => {
        timeLeft.value--
        if (timeLeft.value <= 0) endGame()
      }, 1000)
    })
    onUnmounted(() => { if (_interval) clearInterval(_interval) })
    ```
